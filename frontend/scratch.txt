+"use client";
+
+import { useRef, useState, useEffect } from "react";
+import Webcam from "react-webcam";
+import { useRouter } from "next/navigation";
+import { motion } from "framer-motion";
+import { Camera, CheckCircle2, AlertCircle, Scan, ArrowRight } from "lucide-react";
+
+export default function CameraAssessment() {
+  const webcamRef = useRef<Webcam>(null);
+  const router = useRouter();
+  const [hasPermission, setHasPermission] = useState<boolean | null>(null);
+  const [assessmentState, setAssessmentState] = useState<"idle" | "detecting" | "calibrated" | "recording" | "done">("idle");
+  const [progress, setProgress] = useState(0);
+
+  useEffect(() => {
+    // Request camera permission on mount
+    navigator.mediaDevices.getUserMedia({ video: true })
+      .then(() => setHasPermission(true))
+      .catch(() => setHasPermission(false));
+  }, []);
+
+  const handleStartCalibration = () => {
+    setAssessmentState("detecting");
+    // Mock calibration delay
+    setTimeout(() => {
+      setAssessmentState("calibrated");
+    }, 3000);
+  };
+
+  const handleStartRecording = () => {
+    setAssessmentState("recording");
+    // Mock recording progress
+    const interval = setInterval(() => {
+      setProgress(p => {
+        if (p >= 100) {
+          clearInterval(interval);
+          setAssessmentState("done");
+          setTimeout(() => router.push("/screening/analyze"), 1000);
+          return 100;
+        }
+        return p + 2; // 50 ticks = ~2.5 seconds
+      });
+    }, 50);
+  };
+
+  if (hasPermission === false) {
+    return (
+      <div className="p-8 h-full flex flex-col items-center justify-center">
+        <AlertCircle className="w-12 h-12 text-destructive mb-4" />
+        <h2 className="text-2xl font-semibold mb-2">Camera Access Denied</h2>
+        <p className="text-muted-foreground text-center max-w-md">
+          Doctor AI requires camera access to perform the movement assessment. Please enable camera permissions in your browser settings.
+        </p>
+      </div>
+    );
+  }
+
+  return (
+    <div className="p-8 max-w-5xl mx-auto h-full flex flex-col">
+      <div className="mb-8">
+        <h1 className="text-3xl font-semibold tracking-tight text-white flex items-center gap-3">
+          <Camera className="w-8 h-8 text-accent" />
+          Live Movement Assessment
+        </h1>
+        <p className="text-muted-foreground mt-2">
+          Position yourself so your full body is visible in the frame.
+        </p>
+      </div>
+
+      <div className="flex-1 grid grid-cols-1 lg:grid-cols-3 gap-8">
+        
+        {/* Camera Viewfinder */}
+        <div className="lg:col-span-2 relative rounded-2xl overflow-hidden glass-panel flex flex-col items-center justify-center bg-black/40 border-white/10 h-[500px]">
+          {hasPermission === null ? (
+            <div className="text-white/50 flex flex-col items-center">
+              <Camera className="w-12 h-12 mb-4 animate-pulse" />
+              <p>Requesting camera access...</p>
+            </div>
+          ) : (
+            <>
+              <Webcam
+                ref={webcamRef}
+                audio={false}
+                mirrored={true}
+                className="absolute inset-0 w-full h-full object-cover opacity-60"
+              />
+              
+              {/* Overlay UI */}
+              <div className="absolute inset-0 pointer-events-none p-6 flex flex-col justify-between">
+                <div className="flex justify-between items-start">
+                  <div className="px-3 py-1 bg-black/50 backdrop-blur-md rounded-md border border-white/10 text-xs font-mono text-white/70 flex items-center gap-2">
+                    <span className="w-2 h-2 rounded-full bg-emerald-500 animate-pulse" />
+                    LIVE
+                  </div>
+                  {assessmentState === "detecting" && (
+                    <div className="px-3 py-1 bg-accent/20 text-accent border border-accent/30 rounded-md text-xs font-mono animate-pulse flex items-center gap-2">
+                      <Scan className="w-3 h-3" />
+                      DETECTING LANDMARKS...
+                    </div>
+                  )}
+                </div>
+
+                {assessmentState === "idle" && (
+                  <div className="self-center flex flex-col items-center text-center">
+                    <div className="w-48 h-64 border-2 border-dashed border-white/30 rounded-xl mb-4" />
+                    <p className="text-white font-medium bg-black/50 px-4 py-2 rounded-full backdrop-blur-md">
+                      Stand 1-2 meters away
+                    </p>
+                  </div>
+                )}
+                
+                {assessmentState === "recording" && (
+                  <div className="self-center flex flex-col items-center text-center">
+                    <p className="text-white font-medium bg-destructive/80 px-6 py-3 rounded-full backdrop-blur-md animate-pulse">
+                      Please walk naturally
+                    </p>
+                  </div>
+                )}
+              </div>
+            </>
+          )}
+        </div>
+
+        {/* Instructions Panel */}
+        <div className="flex flex-col gap-6">
+          <div className="glass-panel p-6">
+            <h3 className="font-medium text-white mb-4 uppercase tracking-wider text-sm">Status</h3>
+            
+            <ul className="space-y-4">
+              <li className={`flex items-start gap-3 ${assessmentState !== "idle" ? "text-white" : "text-muted-foreground"}`}>
+                <CheckCircle2 className={`w-5 h-5 mt-0.5 ${assessmentState !== "idle" ? "text-emerald-500" : ""}`} />
+                <div>
+                  <p className="font-medium">Camera connected</p>
+                  <p className="text-xs opacity-70">Video feed active</p>
+                </div>
+              </li>
+              
+              <li className={`flex items-start gap-3 ${assessmentState === "calibrated" || assessmentState === "recording" || assessmentState === "done" ? "text-white" : "text-muted-foreground"}`}>
+                <CheckCircle2 className={`w-5 h-5 mt-0.5 ${assessmentState === "calibrated" || assessmentState === "recording" || assessmentState === "done" ? "text-emerald-500" : ""}`} />
+                <div>
+                  <p className="font-medium">Pose Calibration</p>
+                  <p className="text-xs opacity-70">33 landmarks detected</p>
+                </div>
+              </li>
+
+              <li className={`flex items-start gap-3 ${assessmentState === "done" ? "text-white" : "text-muted-foreground"}`}>
+                <CheckCircle2 className={`w-5 h-5 mt-0.5 ${assessmentState === "done" ? "text-emerald-500" : ""}`} />
+                <div>
+                  <p className="font-medium">Gait Capture</p>
+                  <p className="text-xs opacity-70">Recording complete</p>
+                </div>
+              </li>
+            </ul>
+          </div>
+
+          <div className="glass-panel p-6 flex-1 flex flex-col justify-end">
+            {assessmentState === "idle" && (
+              <button 
+                onClick={handleStartCalibration}
+                className="w-full py-3 bg-white text-black font-medium rounded-lg hover:bg-gray-200 transition-colors"
+              >
+                Calibrate Pose
+              </button>
+            )}
+            
+            {assessmentState === "detecting" && (
+              <button disabled className="w-full py-3 bg-white/10 text-white/50 font-medium rounded-lg cursor-not-allowed">
+                Detecting...
+              </button>
+            )}
+            
+            {assessmentState === "calibrated" && (
+              <button 
+                onClick={handleStartRecording}
+                className="w-full py-3 bg-accent text-white font-medium rounded-lg hover:bg-accent/90 transition-colors"
+              >
+                Start Walking Test
+              </button>
+            )}
+            
+            {assessmentState === "recording" && (
+              <div className="w-full">
+                <div className="flex justify-between text-xs mb-2 text-white/70">
+                  <span>Recording...</span>
+                  <span>{progress}%</span>
+                </div>
+                <div className="w-full h-2 bg-white/10 rounded-full overflow-hidden">
+                  <div 
+                    className="h-full bg-destructive transition-all duration-75"
+                    style={{ width: `${progress}%` }}
+                  />
+                </div>
+              </div>
+            )}
+            
+            {assessmentState === "done" && (
+              <button className="w-full py-3 bg-emerald-500 text-white font-medium rounded-lg flex items-center justify-center gap-2">
+                Processing <ArrowRight className="w-4 h-4 animate-pulse" />
+              </button>
+            )}
+          </div>
+        </div>
+        
+      </div>
+    </div>
+  );
+}
